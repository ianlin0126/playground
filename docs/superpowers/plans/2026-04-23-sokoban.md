# Sokoban Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a single-file browser Sokoban game at `games/sokoban/index.html` — a 🐱 cat pushes 📦 boxes onto 🐟 fish targets across 12 hand-crafted levels, with swipe/keyboard controls and an auto-follow viewport.

**Architecture:** Single self-contained `index.html` (no external deps, all CSS/JS inline). State held in 4 module-level variables (`level`, `undoStack`, `moves`, `won`). Render loop rebuilds the board DOM from scratch on every state change. The viewport auto-follows the cat via `CSS transform: translate()` on an `overflow: hidden` wrapper.

**Tech Stack:** Vanilla HTML/CSS/JS (ES5-compatible), emoji for all visuals, localStorage for progress.

```js
// ── State ──
var level;      // { rows, cols, walls, targets, boxes, playerRow, playerCol }
var undoStack;  // array of { playerRow, playerCol, boxes }
var moves;      // integer
var won;        // boolean
var currentLevelIdx; // 0-based index into LEVELS
var progress;   // array of booleans [false x12], true = completed

var CELL = 46;  // cell size px (44 + 2px gap)
var MAX_UNDO = 200;

// ── Parse a level grid array into working state ──
function parseLevel(levelDef) {
  var grid = levelDef.grid;
  var rows = grid.length;
  var cols = grid[0].length;
  var walls   = {};
  var targets = {};
  var boxes   = [];
  var playerRow = 0, playerCol = 0;

  for (var r = 0; r < rows; r++) {
    for (var c = 0; c < cols; c++) {
      var ch = grid[r][c];
      var key = r + ',' + c;
      if (ch === '#')             walls[key]   = true;
      if (ch === 'X' || ch === '*' || ch === '+') targets[key] = true;
      if (ch === '$' || ch === '*') boxes.push([r, c]);
      if (ch === '@' || ch === '+') { playerRow = r; playerCol = c; }
      // '.' and ' ' are plain floor — no action needed
    }
  }
  return { rows: rows, cols: cols, walls: walls, targets: targets,
           boxes: boxes, playerRow: playerRow, playerCol: playerCol };
}

// ── Load level by 0-based index ──
function loadLevel(idx) {
  currentLevelIdx = idx;
  level = parseLevel(LEVELS[idx]);
  undoStack = [];
  moves = 0;
  won = false;
  render();
  updateHUD();
}
```

- [ ] **Step 2: Add tryMove(dr, dc) after checkWin()**

```js
function tryMove(dr, dc) {
  if (won) return;

  var nr = level.playerRow + dr;
  var nc = level.playerCol + dc;
  var nKey = nr + ',' + nc;

  // Can't move into a wall
  if (level.walls[nKey]) return;
  // Can't move outside the grid
  if (nr < 0 || nr >= level.rows || nc < 0 || nc >= level.cols) return;

  // Check if a box is in the way
  var boxIdx = -1;
  for (var i = 0; i < level.boxes.length; i++) {
    if (level.boxes[i][0] === nr && level.boxes[i][1] === nc) { boxIdx = i; break; }
  }

  if (boxIdx >= 0) {
    // Try to push the box
    var br = nr + dr;
    var bc = nc + dc;
    var bKey = br + ',' + bc;
    // Box is blocked by wall, grid edge, or another box
    if (level.walls[bKey]) return;
    if (br < 0 || br >= level.rows || bc < 0 || bc >= level.cols) return;
    for (var j = 0; j < level.boxes.length; j++) {
      if (j !== boxIdx && level.boxes[j][0] === br && level.boxes[j][1] === bc) return;
    }
    // Save snapshot BEFORE mutating state
    var snapshot = {
      playerRow: level.playerRow, playerCol: level.playerCol,
      boxes: level.boxes.map(function(b) { return [b[0], b[1]]; })
    };
    if (undoStack.length >= MAX_UNDO) undoStack.shift();
    undoStack.push(snapshot);

    // Move box
    level.boxes[boxIdx][0] = br;
    level.boxes[boxIdx][1] = bc;
  } else {
    // No box — save snapshot before moving player
    var snapshot = {
      playerRow: level.playerRow, playerCol: level.playerCol,
      boxes: level.boxes.map(function(b) { return [b[0], b[1]]; })
    };
    if (undoStack.length >= MAX_UNDO) undoStack.shift();
    undoStack.push(snapshot);
  }

  // Move player
  level.playerRow = nr;
  level.playerCol = nc;
  moves++;

  render();
  updateHUD();

  if (checkWin()) {
    won = true;
    progress[currentLevelIdx] = true;
    saveProgress();
    showWinScreen();
  }
}
```

function undo() {
  if (won || undoStack.length === 0) return;
  var snap = undoStack.pop();
  level.playerRow = snap.playerRow;
  level.playerCol = snap.playerCol;
  level.boxes = snap.boxes.map(function(b) { return [b[0], b[1]]; });

  render();
  updateHUD();
}

function restart() {

// ── Swipe controls ──
var touchStartX = null, touchStartY = null;
var boardWrap = document.getElementById('board-wrap');

boardWrap.addEventListener('touchstart', function(e) {
  touchStartX = e.touches[0].clientX;
  touchStartY = e.touches[0].clientY;
}, { passive: true });

boardWrap.addEventListener('touchend', function(e) {
  if (touchStartX === null) return;
  var dx = e.changedTouches[0].clientX - touchStartX;
  var dy = e.changedTouches[0].clientY - touchStartY;
  touchStartX = null;
  touchStartY = null;

  var minSwipe = 20;  // px threshold to count as a swipe
  if (Math.abs(dx) < minSwipe && Math.abs(dy) < minSwipe) return;

  if (Math.abs(dx) >= Math.abs(dy)) {

function buildLevelSelect() {
  var grid = document.getElementById('level-grid');
  grid.innerHTML = '';
  for (var i = 0; i < LEVELS.length; i++) {
    var btn = document.createElement('button');
    btn.className = 'lvl-btn';
    var numSpan  = document.createElement('span');
    numSpan.className = 'lvl-num';
    numSpan.textContent = LEVELS[i].id;
    var iconSpan = document.createElement('span');

    if (progress[i]) {
      btn.classList.add('done');
      iconSpan.textContent = '✅';
    } else if (i === currentLevelIdx) {
      btn.classList.add('current');
      iconSpan.textContent = '🐱';
    } else if (i === 0 || progress[i - 1]) {
      // Unlocked: first level is always unlocked; others unlock when previous is complete
      btn.classList.add(i === currentLevelIdx ? 'current' : '');
      iconSpan.textContent = '';
    } else {
      btn.classList.add('locked');
      iconSpan.textContent = '🔒';
      btn.disabled = true;
    }

    btn.appendChild(numSpan);
    btn.appendChild(iconSpan);

    if (!btn.disabled) {
      (function(idx) {
        btn.addEventListener('click', function() {
          document.getElementById('level-overlay').classList.remove('show');
          loadLevel(idx);
        });
      }(i));
    }

    grid.appendChild(btn);
  }
}
